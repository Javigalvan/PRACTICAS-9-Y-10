package ejemplo;

public class Rectangulo {
	
	private double base;
    private double altura;

    public Rectangulo(double base, double altura) {
        this.base = base;
        this.altura = altura;
    }

    public double area() {

        // Valores negativos
        if (base < 0 || altura < 0) {
            return -1;
        }

        // Valores cero
        if (base == 0 || altura == 0) {
            return 0;
        }

        // Valores positivos
        return base * altura;
    }

    public double perimetro() {

        // Valores negativos
        if (base < 0 || altura < 0) {
            return -1;
        }

        // Valores cero
        if (base == 0 || altura == 0) {
            return 0;
        }

        // Valores positivos
        return 2 * (base + altura);
    }
}
