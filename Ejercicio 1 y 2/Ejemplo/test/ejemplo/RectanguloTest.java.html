        Rectangulo r = new Rectangulo(5, 3);
        assertEquals(15, r.area());
    }

    @Test

        Rectangulo r = new Rectangulo(5, 3);
        assertEquals(16, r.perimetro());
    }

    // Valores negativos

        Rectangulo r = new Rectangulo(-5, 3);
        assertEquals(-1, r.area());

        Rectangulo r = new Rectangulo(5, -3);
        assertEquals(-1, r.perimetro());
    }

    // Valores cero
